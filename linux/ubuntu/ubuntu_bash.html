#!/bin/bash
# shellcheck disable=SC2162

# 1. Prompt the user for input using 'read' (-p allows adding a prompt message inline)
read -p "Enter your name: " user_name
read -p "Enter your age: " user_age

echo "----------------------------------------"
echo "Hello, $user_name! Let's check your category."

# 2. Conditional statement (if-elif-else)
# In Bash, numeric comparisons use flags like:
# -ge (greater than or equal)
# -lt (less than)
# -eq (equal to)
# Note the spaces inside the double brackets [[ ... ]] !

if [[ $user_age -ge 18 ]]; then
    echo "Status: You are an adult."
elif [[ $user_age -ge 13 ]]; then
    echo "Status: You are a teenager."
else
    echo "Status: You are a child."
fi

echo "----------------------------------------"
  
chmod +x hello.sh
bash hello.sh
