Connect to MySQL Shell

mysql -h localhost -P 4306 -u root -p
show databases;

Export Database

mysqldump -h localhost -P 4306 -u root -p source_database_name  > "C:\Users\DELL\Desktop\destination_database_name.sql"

Export Data-Only

mysqldump -h localhost -P 4306 -u root -p  --no-create-info source_database_name > "C:\Users\DELL\Desktop\destination_database_name.sql"

Export Schema-Only

mysqldump -h localhost -P 4306 -u root -p  --no-data --routines --triggers --events source_database_name > "C:\Users\DELL\Desktop\destination_database_name.sql"

Restore Database

Create an empty database first
mysql -h localhost -P 4306 -u root -p -e "CREATE DATABASE destination_database_name;"
Restore the database from sql file
mysql -h localhost -P 4306 -u root -p destination_database_name < "C:\Users\DELL\Desktop\source_database_name.sql"
